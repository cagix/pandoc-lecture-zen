\newcommand{\DeclareHSBIAlertBox}[5]{%
  %% #1 name, #2 default title, #3 frame colour, #4 background colour, #5 title background colour
  \newtcolorbox{#1}[1][#2]{%
    hsbi-alert,
    colframe=#3,
    colback=#4,
    colbacktitle=#5,
    title={##1},
  }%
}
