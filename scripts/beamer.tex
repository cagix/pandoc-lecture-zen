\newcommand{\DeclareHSBIBeamerBox}[5]{%
  %% #1 name, #2 default title, #3 title colour, #4 background colour, #5 title text colour
  \newtcolorbox{#1}[1][#2]{%
    hsbi-beamerblock,
    colback=#4,
    coltitle=#5,
    colbacktitle=#3,
    title={##1},
  }%
}
